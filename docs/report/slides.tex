\begin{frame}{مدل در یک اسلاید}
\small
$\Gamma=\langle N,\Theta,p,E,T,H,M,A,\Sigma,D,S,g,G,\varphi,B,X,\mu,\tau,u\rangle$
\begin{itemize}\itemsep2pt
  \item نوع خصوصی $\theta_i=(\rbar_i,\mbar_i)$: عزم پایه و ظرفیت مدیریت
        هزینه؛ ثابت و \emph{مشاهده‌نشدنی}.
  \item تاب‌آوری $e_i^t$: حالت \emph{نهفته}، با
        $e^{t+1}=e^t-g^t+\xi$؛ رسیدن به صفر $\Rightarrow$ خروج اجباری.
  \item اقدام $a_i^t=(\sig_i^t,d_i^t)$، $\sig\in\{0,.25,.5,.75,1\}$.
  \item هزینه‌ی مؤثر
        $g_i=\dfrac{c_i+\alpha_i K_i+r_i}{\mbar_i(0.35+0.65\ehat_i)}$؛
        $K$ = انبار تعهد عمومی.
  \item دو ارزش پایانی:
        $B_i=175(0.5+0.3\ehat_i+0.2\rbar_i)$ و
        $X_i=x_0-5K_i-8\tilde\sig_j+20M$.
\end{itemize}
\vfill
\begin{alertblock}{اتحاد کلیدی}
$G_i(t)\approx e_i^0-e_i^t$ — «هزینه‌ی انباشته» و «تاب‌آوری خرج‌شده» یک چیزند.
\end{alertblock}
\end{frame}

\begin{frame}{جای این بازی در نقشه‌ی درس}
\begin{itemize}\itemsep4pt
  \item \textbf{بازی زمان‌بندی (جلسه‌ی ۱۱)} — از نوع \emph{فرسایشی}، نه
        پیش‌دستی: بیشتر ماندنِ من تاب‌آوری حریف را می‌سوزاند، پس انگیزه
        دیرتر رفتن است.
  \item \textbf{نسخه‌ی بیزی با نوع پیوسته (جلسه‌ی ۷)} — استراتژی تعادلی
        نسبت به نوع صعودی است؛ نسخه‌ی گسسته‌اش همان ساختار آستانه‌ای ماست.
  \item \textbf{سیگنال‌دهی و \lr{PBE} (جلسه‌ی ۱۲)} — مثال اسپنس فرض
        می‌کرد هزینه‌ی سیگنال برای نوع \emph{ضعیف} بیشتر است؛ این همان
        شرط تک‌تقاطعی روی $\varphi_i(\sig)$ است.
  \item \textbf{استراتژی رفتاری (جلسه‌ی ۱۰)} — سیاست عامل دقیقاً یک توزیع
        روی اکشن‌ها در هر مجموعه‌ی اطلاعاتی است.
\end{itemize}
\vfill
\begin{alertblock}{ولی سه جا از مدل‌های درس فراتر می‌رود}
\centering\small
سیگنال‌دهی \emph{دوطرفه} است (هر دو هم فرستنده‌اند هم گیرنده)؛ سیگنال
\emph{پنج‌سطحی} است نه دودویی؛ و ارزش خروج $X$ \emph{درون‌زا}ست.
\end{alertblock}
\end{frame}

\begin{frame}{۱. دروازه‌ی ادامه}
هزینه‌های گذشته ریخته‌اند، پس مقایسه آینده‌نگر است:
\[
  \boxed{\;q_i^t\ \ge\ q^\ast=\frac{g_i+\varphi_i(\sig_i)-\Delta X_i}{B_i-X_i}\;}
\]
\begin{itemize}\itemsep2pt
  \item \textbf{صورت:} سوخت + بهای سیگنال + \emph{فرسایش گزینه‌ی خروج}
        ($-\Delta X$).
  \item \textbf{مخرج:} فاصله‌ی جایزه تا گزینه‌ی بیرونی.
\end{itemize}
\vfill
حالا عددهای واقعی این جهان:
\[
  B\in[87.5,\,185.5],\qquad X\in[-2,\,48],\qquad g\approx1.1\text{–}1.3
\]
\[
  \Rightarrow\qquad q^\ast\approx\frac{1.12}{150-26}\approx\mathbf{0.009}
\]
\begin{alertblock}{یافته‌ی ۱}
خروج \emph{داوطلبانه} تقریباً هیچ‌وقت بهینه نیست. خروج عقلانی همان خروج
\emph{اجباری} است.
\end{alertblock}
\end{frame}

% ---------------------------------------------------------------- %
\begin{frame}{۲. باورها: مسئله‌ی فیلتر فقط دوبعدی است}
حالت $e_j^t$ پنهان است، ولی \emph{قانونش} عمومی است:
\[
  g_j^t=\mathrm{ramp}(t)\cdot
  \frac{c_{0j}+\kappa_j\sig_i+\alpha_j K_j+\eta\sig_i\sig_j}
       {\mbar_j\,(0.35+0.65\,e_j/e_j^0)}
\]
هر چیزی جز $(\mbar_j,e_j^0)$ عمومی است $\Rightarrow$
\textbf{پسین روی \emph{مسیر} = پسین روی یک \emph{نوع دوبعدی}}.
\vfill
سه کانال به‌روزرسانی:
\begin{enumerate}\itemsep2pt
  \item \textbf{بقا:} حریف هنوز نرفته $\Rightarrow$ هر نوعی که باید مرده
        بود حذف می‌شود. \emph{دقیق}، بدون نیاز به مدل استراتژی حریف.
  \item \textbf{تک‌تقاطعی:} موضع بلندِ انتخاب‌شده شاهد قوی بودن است.
  \item پیشین منتشرشده (نامتقارن).
\end{enumerate}
\end{frame}

\begin{frame}{۳. قوی‌ترین سیگنال بهترین نیست}
تعارض اعتبار--انعطاف از سه کانال می‌گذرد: انبار تعهد ($-5K$)، تحقیر
($-8\tilde\sig_j$)، و \emph{قفل نوبت} ($\sig\ge0.75$ $\Rightarrow$ مجبور به
\lr{HOLD}).
\vfill
\begin{alertblock}{یافته‌ی ۲ — تخفیفی که موتور به تشدید می‌دهد}
موتور $\varphi$ را فقط وقتی سیگنال \emph{تازه} تولید شود می‌گیرد؛ در
\lr{HOLD} \textbf{هیچ هزینه‌ای گرفته نمی‌شود}. و عبور از $0.75$ دقیقاً همان
چیزی است که شما را \lr{HOLD} می‌کند.
\[
  \tfrac12\varphi(0.75)=0.281\kappa
  \quad\text{در برابر}\quad
  \varphi(0.50)=0.25\kappa
\]
هزینه تقریباً برابر، ولی فشار وارد بر حریف به‌مراتب بیشتر.
\end{alertblock}
\centering
$\Rightarrow$ تابع سود نسبت به $\sig$ \textbf{یکنوا نیست}؛ $\sig=0.5$
بدترین گزینه است.
\end{frame}

\begin{frame}{دو تصحیح، و یکی‌شان کل نظریه را ساده کرد}
\begin{enumerate}\itemsep6pt
  \item \textbf{استخر آموزش:} از $\{$\lr{Random}, \lr{TitForTat}$\}$ به
        همان شش حریف برد. عامل ما هیچ‌وقت با حریفی که \emph{نمی‌رود}
        تمرین نکرده بود، و مقابل \lr{Hawk} در صندلی آمریکا $-58.7$
        می‌گرفت؛ حالا $+4.7$.
  \item \textbf{جمله‌ی جریمه‌ی تسلیم حذف شد.} امتیاز \lr{TitForTat} از
        بازی‌هایش با \emph{استخر مرجع} می‌آید و ما عضو آن استخر نیستیم
        $\Rightarrow$ عدد $53.8$ او ثابت است و ما تکانش نمی‌دهیم.
\end{enumerate}
\begin{block}{یعنی معیار اصلاً نسبی نیست}
\centering
میانگین مطلوبیت خودمان روی استخر مرجع $>\ 53.8$
\end{block}
\small
پس جمله‌ی محروم‌سازی فقط هزینه بود بدون عایدی. با حذفش شکل‌دهی پاداش دوباره
\emph{کاملاً پتانسیل‌پایه} شد $\Rightarrow$ طبق
\lr{Ng, Harada \& Russell} سیاست بهینه را عوض نمی‌کند، و عامل دقیقاً همان
کمیتی را بیشینه می‌کند که برد می‌سنجد.
\end{frame}

\begin{frame}{یافته‌ی ۱۱: برون‌یابی را با درون‌یابی اشتباه نگیرید}
برآوردگر ما برای آپلود سوم $64.8$ گفت، برد $58.8$ داد. ولی خودش کنار
عددش نوشته بود:
\begin{alertblock}{}
\centering\small
«این کاندیدا $27.9$ واحد بیرون از بازه‌ی لنگرهاست $\Rightarrow$
\emph{برون‌یابی} است نه درون‌یابی. یک \textbf{جهت} بدانید، نه یک
\textbf{عدد}.»
\end{alertblock}
جهت درست بود ($+2.0$ واقعی)، عدد غلط ($+8.0$ ادعا شده). با لنگر سوم
معلوم شد رابطه اشباع می‌شود:

\end{frame}

\begin{frame}{یافته‌ی ۱۰: استخر را جایگزین کردیم، نه گسترش}
\begin{alertblock}{}
\centering
\lr{--pool board} کل دیکشنری حریف‌ها را عوض کرد $\Rightarrow$
\lr{Patient} و \lr{Plateau} بی‌سروصدا افتادند.
\end{alertblock}
عامل یاد گرفت تسلیم شدن هزینه‌ای ندارد، چون دیگر هیچ‌کس نبود که فقط
بایستد و صبر کند. بهایش روی برد $\approx13$ واحد بود.
\begin{block}{درسِ عملی}
این دسته خطا را با \emph{امتیاز} پیدا نمی‌کنید --- امتیاز کل بالا رفت.
با اندازه‌گیری \emph{رفتار} پیدا می‌کنید.
\end{block}
\small
و وقتی بازسازی جواب نمی‌دهد، دنبال \emph{جانشینِ هم‌ترتیب} بگردید:
\lr{Patient} تنها حریف محلی است که دو آپلود ما را به همان ترتیبِ برد
می‌چیند ($-18.6/-43.3$ در برابر $-39.5/-71.3$). سطحش شبیه نیست و ادعایی
هم نمی‌کنیم --- برای \emph{انتخاب} ترتیب کافی است.
\end{frame}

\begin{frame}{یافته‌ی ۱۴: تمام این مدت، جدولِ اشتباه را بهینه می‌کردیم}

به‌علاوه‌ی $20\%$ نمره‌ی اضافه برای رتبه در همان تورنمنت.

\begin{alertblock}{}
\centering\small
پس صدرنشینیِ جدول تمرینی هیچ تضمینی برای معیار نمره‌دار نیست — دو
میدان متفاوت‌اند و باید جداگانه سنجیده شوند.
\end{alertblock}

\begin{frame}{جمع‌بندی: سه حلقه‌ی خودتقویت‌کننده}
\begin{enumerate}\itemsep6pt
  \item \textbf{حلقه‌ی تعهد:} تشدید $\Rightarrow K\uparrow \Rightarrow
        X\downarrow \Rightarrow (B-X)\uparrow \Rightarrow q^\ast\downarrow
        \Rightarrow$ ادامه عقلانی‌تر.
  \item \textbf{حلقه‌ی شکنندگی:} فرسایش $\Rightarrow m\downarrow \Rightarrow
        g\uparrow \Rightarrow$ فرسایش سریع‌تر.
  \item \textbf{حلقه‌ی میانجی:} تشدید $\Rightarrow$ پنجره بسته
        $\Rightarrow X$ پایین $\Rightarrow$ خروج غیرممکن‌تر.
\end{enumerate}
\vfill
\begin{block}{و تنها چیزی که هر سه را می‌شکند}
\centering
\emph{خویشتن‌داری متقابل} — چون هم‌زمان $K$ را می‌خواباند، سوخت را کم می‌کند،
و پنجره‌ی میانجی‌گری را باز می‌کند.
\end{block}
\vfill
\centering\small
محدودیت اصلی: مدل درباره‌ی \emph{جهت} روابط قابل دفاع است، نه درباره‌ی
\emph{اندازه}شان.
\end{frame}
